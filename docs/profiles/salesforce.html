profiles/salesforce.conf

# salesforce: the Salesforce CLI on a managed Node, plus daily-use aliases.
summary: "Salesforce CLI (sf) on a managed Node LTS, with helper aliases"
include: node

cmd salesforce-cli run="fnm exec --using=lts-latest npm install -g @salesforce/cli" check="fnm exec --using=lts-latest sf --version" why="Installs sf into the fnm-managed Node"
tool sf min=2.0 why="Visible once fnm's shell hook is loaded"
block sf-aliases file="~/.zshrc" line="alias sfo='sf org open'" line="alias sfls='sf org list'" line="alias sfdeploy='sf project deploy start'" line="alias sfpull='sf project retrieve start'" why="Short names for the commands you type all day"
