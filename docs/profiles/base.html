Manifest

profiles/base.conf

# base: the command-line tools every machine gets.
summary: "Core command-line tools: git, gh, jq, ripgrep, fd, fzf, bat, shellcheck, tmux"

# Make Homebrew available in every login shell (Apple Silicon and Intel prefixes).
block homebrew-shellenv file="~/.zprofile" line="if [ -x /opt/homebrew/bin/brew ]; then eval \"$(/opt/homebrew/bin/brew shellenv)\"; elif [ -x /usr/local/bin/brew ]; then eval \"$(/usr/local/bin/brew shellenv)\"; fi" why="Puts Homebrew's bin directory ahead of /usr/bin"

formula git min=2.30 why="Apple's bundled git lags behind"
formula gh min=2.40 why="GitHub CLI"
formula jq min=1.6
formula ripgrep bin=rg min=13.0 why="Fast recursive search"
formula fd min=9.0 why="Friendlier find"
formula fzf min=0.40 why="Fuzzy finder"
formula bat min=0.22 why="cat with syntax highlighting"
formula shellcheck min=0.9 why="Lints shell scripts"
formula tmux min=3.2 versionarg=-V
