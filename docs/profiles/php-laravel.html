Manifest

profiles/php-laravel.conf

# php-laravel: PHP, Composer, MySQL and Laravel Valet.
#
# Valet runs nginx and dnsmasq as root services so that http://app.test works
# without ports. `valet install` therefore asks for your sudo password; that is
# Valet's own behaviour, not something devbox adds.
summary: "PHP, Composer, MySQL and Laravel Valet"
include: base

formula php min=8.2 why="Homebrew PHP; /usr/bin must not shadow it"
formula composer min=2.5
formula mysql min=8.0 why="Local database server; devbox does not start it"
tool sqlite3 min=3.35 why="Ships with macOS; Laravel's default local database"

block composer-global-bin file="~/.zshrc" line="export PATH=\"$HOME/.composer/vendor/bin:$HOME/.config/composer/vendor/bin:$PATH\"" why="Finds the valet command once Composer has installed it"

cmd valet-composer-require run="composer global require laravel/valet" check="composer global show laravel/valet" why="Installs the Valet command"
cmd valet-install run="export PATH=\"$HOME/.composer/vendor/bin:$HOME/.config/composer/vendor/bin:$PATH\"; valet install" check="test -f \"$HOME/.config/valet/config.json\"" sudo=yes why="Configures dnsmasq, nginx and php-fpm; prompts for sudo"
cmd valet-park run="export PATH=\"$HOME/.composer/vendor/bin:$HOME/.config/composer/vendor/bin:$PATH\"; mkdir -p \"$HOME/Sites\" && cd \"$HOME/Sites\" && valet park" check="export PATH=\"$HOME/.composer/vendor/bin:$HOME/.config/composer/vendor/bin:$PATH\"; valet paths | grep -q \"$HOME/Sites\"" why="Serves every folder in ~/Sites as <folder>.test"

# Expectations that doctor verifies. Valet's install step starts these services.
tool valet min=4.0 why="Needs the composer-global-bin block loaded in your shell"
service dnsmasq why="resolves *.test to 127.0.0.1; valet install starts it"
service nginx why="valet install starts it"
service php why="php-fpm for the Homebrew PHP; valet install starts it"
