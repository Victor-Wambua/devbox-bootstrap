Manifest

profiles/macos-defaults.conf

# macos-defaults: a small, curated and reversible set of macOS preferences.
#
# Before each value is changed, devbox records the previous one in
# ~/.devbox/backups/<time>/restore-defaults.sh, a script you can run to undo it.
summary: "Curated, reversible macOS preferences (Finder, Dock, keyboard, text)"

defaults finder-show-extensions domain=NSGlobalDomain key=AppleShowAllExtensions type=bool value=true why="Show every file extension"
defaults finder-path-bar domain=com.apple.finder key=ShowPathbar type=bool value=true restart=Finder why="Path bar at the bottom of Finder windows"
defaults finder-status-bar domain=com.apple.finder key=ShowStatusBar type=bool value=true restart=Finder
defaults finder-list-view domain=com.apple.finder key=FXPreferredViewStyle type=string value=Nlsv restart=Finder why="List view by default"
defaults finder-no-network-ds-store domain=com.apple.desktopservices key=DSDontWriteNetworkStores type=bool value=true why="No .DS_Store litter on network shares"
defaults dock-autohide domain=com.apple.dock key=autohide type=bool value=true restart=Dock
defaults dock-no-recents domain=com.apple.dock key=show-recents type=bool value=false restart=Dock
defaults keyboard-repeat domain=NSGlobalDomain key=KeyRepeat type=int value=2 why="Faster key repeat; applies after you log out and in"
defaults keyboard-initial-delay domain=NSGlobalDomain key=InitialKeyRepeat type=int value=25
defaults no-smart-quotes domain=NSGlobalDomain key=NSAutomaticQuoteSubstitutionEnabled type=bool value=false why="Curly quotes break pasted code"
defaults no-smart-dashes domain=NSGlobalDomain key=NSAutomaticDashSubstitutionEnabled type=bool value=false
